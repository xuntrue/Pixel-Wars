% \unit{Name}{Damage}{Move speed}{Attack speed}{Health}
% Starts a new section for the unit and prints its catalog stats.
\newcommand{\unit}[5]{%
  \section{#1}\label{unit:#1}
  \begin{center}
    \begin{tabular}{cccc}
      \toprule
      Damage & Move speed & Attack speed & Health \\
      \midrule
      #2 & #3 & #4 & #5 \\
      \bottomrule
    \end{tabular}
  \end{center}
}

% \unitdata{Battles}{Win rate}{95\% CI}{Survival}{Damage/unit}
% Paste the numbers from the analysis report (unit performance section).
\newcommand{\unitdata}[5]{%
  \subsection*{Data summary}
  \begin{center}
    \begin{tabular}{ccccc}
      \toprule
      Battles & Win rate & 95\% CI & Survival & Damage/unit \\
      \midrule
      #1 & #2 & #3 & #4 & #5 \\
      \bottomrule
    \end{tabular}
  \end{center}
}

% \unitref{Name} links to another unit's section (the name must match exactly).
\newcommand{\unitref}[1]{\hyperref[unit:#1]{#1}}
